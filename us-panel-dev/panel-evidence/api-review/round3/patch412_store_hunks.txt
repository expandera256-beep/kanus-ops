+func commitGraceFromEnv() time.Duration {
+	raw := strings.TrimSpace(os.Getenv("US_CREDIT_COMMIT_GRACE_SECONDS"))
+	if raw == "" {
+		return 10 * time.Minute
+	}
+	n, err := strconv.Atoi(raw)
