import("net/http/httptest";"net/http";"strings";"os";"path/filepath";"testing";"net/url")
func TestR2IndependentOriginAndBounds(t *testing.T){
 t.Setenv("US_DESK",filepath.Join(t.TempDir(),"desk.json"));s:=newView("r2");s.root=t.TempDir();h:=guiMux(&s)
 post:=func(path,origin,body string)*httptest.ResponseRecorder{r:=httptest.NewRequest("POST","http://127.0.0.1:4477"+path,strings.NewReader(body));if origin!=""{r.Header.Set("Origin",origin)};r.Header.Set("Content-Type","application/json");w:=httptest.NewRecorder();h.ServeHTTP(w,r);return w}
 for _,o:=range []string{"","https://evil.example","http://127.0.0.1:4477","null","http://localhost:4477","http://127.0.0.1:4478"}{r:=httptest.NewRequest("POST","http://127.0.0.1:4477/api/lang",nil);r.Header.Set("Origin",o);old:=true;if o!=""{u,e:=url.Parse(o);old=e==nil&&u.Scheme=="http"&&strings.EqualFold(u.Host,r.Host)};if sameOriginRequest(r)!=old{t.Fatal("origin changed",o)};want:=200;if !old{want=403};w:=post("/api/lang",o,`{"lang":"en"}`);if w.Code!=want{t.Fatalf("origin=%q code=%d",o,w.Code)};t.Logf("Origin=%q lang=%d legacy-guard=%t",o,w.Code,old);if !old&&post("/api/act",o,`{"op":"refresh"}`).Code!=403{t.Fatal("act foreign origin")}}
 raw:=`{"lang":"en"}`;cases:=[]struct{body string;want int}{{raw+" xxx",400},{raw+`{"lang":"tr"}`,400},{raw+strings.Repeat(" ",2<<20),413},{raw+strings.Repeat(" ",4096-len(raw)),200},{raw+strings.Repeat(" ",4097-len(raw)),413},{raw+strings.Repeat("x",2<<20),400}}
 for _,c:=range cases{w:=post("/api/lang","",c.body);t.Logf("HTTP body size=%d got=%d want=%d",len(c.body),w.Code,c.want);if w.Code!=c.want{t.Error(w.Code,c.want)}}
}
