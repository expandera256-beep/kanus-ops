import("os";"path/filepath";"strings";"testing";"syscall";"time";"sync/atomic")
func TestR2IndependentReadAndLock(t *testing.T){
 d:=t.TempDir();path:=filepath.Join(d,"lang.json");p:=store(path)
 raw:=`{"lang":"en"}`;for _,n:=range []int{len(raw),4096,4097}{os.WriteFile(path,[]byte(raw+strings.Repeat(" ",n-len(raw))),0600);got:=p.Get();want:="en";if n>4096{want=""};if got!=want{t.Fatalf("size=%d got=%q want=%q",n,got,want)};t.Logf("regular size=%d value=%q",n,got)}
 os.WriteFile(path,[]byte(raw),0600);p.mu.Lock();done:=make(chan string,1);go func(){done<-p.Get()}();select{case got:=<-done:if got!="en"{t.Error(got)};case <-time.After(time.Second):t.Error("Get acquired write mutex")};p.mu.Unlock()
 os.Remove(path);syscall.Mkfifo(path,0600);start:=time.Now();if p.Get()!=""{t.Fatal("FIFO accepted")};t.Logf("FIFO rejected elapsed=%s",time.Since(start));os.Remove(path)
 out:=filepath.Join(t.TempDir(),"victim");os.WriteFile(out,[]byte(raw),0600);os.Symlink(out,path);if p.Get()!=""{t.Fatal("symlink followed")};os.Remove(path)
 f,_:=os.Create(path);f.Truncate(1<<30);f.Close();start=time.Now();if p.Get()!=""{t.Fatal("large accepted")};t.Logf("1GiB sparse rejected elapsed=%s",time.Since(start))
}
func TestR2IndependentParentRace(t *testing.T){
 root,out:=t.TempDir(),t.TempDir();slot:=filepath.Join(root,"slot");held:=filepath.Join(root,"held");os.Mkdir(slot,0700);p:=store(filepath.Join(slot,"lang.json"));if err:=p.Set("en");err!=nil{t.Fatal(err)}
 stop:=make(chan struct{});done:=make(chan struct{});var swaps atomic.Int64
 go func(){defer close(done);for{select{case <-stop:return;default:};if os.Rename(slot,held)!=nil{continue};if os.Symlink(out,slot)==nil{swaps.Add(1)};if f,e:=os.Lstat(slot);e==nil&&f.Mode()&os.ModeSymlink!=0{os.Remove(slot)};os.Rename(held,slot)}}()
 ok,bad:=0,0;for i:=0;i<1500;i++{if p.Set("en")==nil{ok++}else{bad++};p.Get()};close(stop);<-done
 entries,_:=os.ReadDir(out);if len(entries)!=0{t.Fatalf("outside writes=%v",entries)};t.Logf("parent-swap races=%d Set success=%d reject=%d outside entries=0",swaps.Load(),ok,bad)
}
